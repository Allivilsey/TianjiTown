package org.allivlisey.tianjitown.integrations.jobs;

import org.bukkit.Server;
import org.bukkit.plugin.Plugin;
import org.bukkit.scheduler.BukkitScheduler;
import org.junit.jupiter.api.Test;
import java.util.concurrent.*;
import java.util.concurrent.atomic.*;
import java.lang.reflect.InvocationTargetException;
import static org.mockito.Mockito.*;
import static org.mockito.ArgumentMatchers.*;
import static org.junit.jupiter.api.Assertions.*;

class AuditJobsTimeoutTest {
    @Test void timedOutProcessorStillExecutesLater() throws Exception {
        Plugin owner = mock(Plugin.class), jobs = mock(Plugin.class);
        Server server = mock(Server.class);
        BukkitScheduler scheduler = mock(BukkitScheduler.class);
        when(owner.getServer()).thenReturn(server);
        when(server.getScheduler()).thenReturn(scheduler);
        AtomicReference<Callable<Object>> queued = new AtomicReference<>();
        Future<Object> future = mock(Future.class);
        when(future.get(10, TimeUnit.SECONDS)).thenThrow(new TimeoutException());
        when(scheduler.callSyncMethod(eq(owner), any())).thenAnswer(call -> {
            queued.set(call.getArgument(1));
            return future;
        });
        AtomicInteger settlements = new AtomicInteger();
        var adapter = new JobsIncomeTaxAdapter(owner, jobs, () -> true, earning -> {
            settlements.incrementAndGet();
            return JobsIncomeTaxAdapter.TaxResult.taxed(95);
        }, (key, args) -> key);
        var process = JobsIncomeTaxAdapter.class.getDeclaredMethod("process", JobsIncomeTaxAdapter.Earning.class);
        process.setAccessible(true);
        assertThrows(InvocationTargetException.class, () -> process.invoke(adapter, new JobsIncomeTaxAdapter.Earning(null, 100)));
        assertEquals(0, settlements.get());
        queued.get().call();
        assertEquals(1, settlements.get(), "BUG: settlement runs after event processing timed out");
        verify(future, never()).cancel(anyBoolean());
    }
}
